-- Connect to your PostgreSQL database and run this SQL: psql -d minio_events

-- Step 1: Create a trigger function
CREATE OR REPLACE FUNCTION notify_new_entry() RETURNS TRIGGER AS $$
BEGIN
  -- Notify channel "new_entry_channel" with the inserted row's ID
  PERFORM pg_notify('new_entry_channel', NEW.key::text);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Step 2: Attach trigger to your table (e.g., "events")
CREATE TRIGGER new_entry_trigger
AFTER INSERT ON events
FOR EACH ROW
EXECUTE FUNCTION notify_new_entry();
